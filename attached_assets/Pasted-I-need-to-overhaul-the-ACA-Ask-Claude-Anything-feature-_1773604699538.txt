async function runWithOpenAICompatible(
  provider: "grok" | "openai",
  messages: AcaMessage[],
  clientContext?: ClientContext,
  integrations?: string[]
): Promise<string> {
  // 1. Build the OpenAI client
  let client: OpenAI;
  if (provider === "grok") {
    client = new OpenAI({ apiKey: process.env.GROQ_API_KEY, baseURL: "https://api.x.ai/v1" });
  } else {
    client = new OpenAI({ apiKey: process.env.OPENAI_API_KEY });
  }
  const model = provider === "grok"
    ? (process.env.GROK_MODEL || "grok-3-latest")
    : (process.env.OPENAI_MODEL || "gpt-4o");

  // 2. Pre-fetch data from selected integrations to inject into prompt
  //    This avoids tool-use (which Grok/OpenAI handle differently) and saves tokens
  //    by only fetching what the user explicitly selected.
  let dataContext = "";
  if (clientContext && integrations && integrations.length > 0) {
    const dataBlocks: string[] = [];
    for (const integration of integrations) {
      const toolNames = INTEGRATION_TO_TOOLS[integration];
      if (!toolNames) continue;
      for (const toolName of toolNames) {
        try {
          // Build a default input for each tool
          const toolInput: Record<string, any> = { client_id: clientContext.id };
          if (integration === "nsm_goals") {
            toolInput.client_name = clientContext.name;
            delete toolInput.client_id;
          }
          // Use a sensible default command for tools that require one
          const defaultCommands: Record<string, string> = {
            query_google_search_console: "gsc_top_queries",
            query_google_analytics: "ga4_combined_funnel",
            query_callrail: "callrail_summary",
            query_ctm: "ctm_qoq_organic_calls",
            query_semrush: "semrush_organic_overview",
            query_ahrefs: "ahrefs_backlink_overview",
            query_screaming_frog: "technical_health_summary",
          };
          if (defaultCommands[toolName]) {
            toolInput.command = defaultCommands[toolName];
          }
          console.log(`[ACA][${provider}] Pre-fetching: ${toolName}`);
          const result = await executeTool(toolName, toolInput);
          dataBlocks.push(`### ${toolName}\n${result}`);
        } catch (err: any) {
          dataBlocks.push(`### ${toolName}\nError: ${err.message}`);
        }
      }
    }
    if (dataBlocks.length > 0) {
      dataContext = "\n\nDATA CONTEXT (pre-fetched from selected integrations):\n" + dataBlocks.join("\n\n");
    }
  }

  // 3. Build system prompt with data baked in
  const systemPrompt = buildSystemPrompt(clientContext, integrations) + dataContext;

  // 4. Make a single chat completion call (no tool loop)
  const openaiMessages: Array<{ role: "system" | "user" | "assistant"; content: string }> = [
    { role: "system", content: systemPrompt },
    ...messages.map(m => ({ role: m.role as "user" | "assistant", content: m.content })),
  ];

  const completion = await client.chat.completions.create({
    model,
    messages: openaiMessages,
    max_tokens: 4096,
  });

  return completion.choices[0]?.message?.content || "No response generated.";
}

function getFilteredTools(integrations?: string[]): Anthropic.Tool[] {
  if (!integrations || integrations.length === 0) return ACA_TOOLS; // backward compatible
  const allowedToolNames = new Set(ALWAYS_AVAILABLE_TOOLS);
  for (const key of integrations) {
    const tools = INTEGRATION_TO_TOOLS[key];
    if (tools) tools.forEach(t => allowedToolNames.add(t));
  }
  return ACA_TOOLS.filter(t => allowedToolNames.has(t.name));
}
